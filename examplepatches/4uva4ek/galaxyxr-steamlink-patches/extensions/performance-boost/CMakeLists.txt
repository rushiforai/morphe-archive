cmake_minimum_required(VERSION 3.22)
project(gxr_performance_boost LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(openxr_headers GIT_REPOSITORY https://github.com/KhronosGroup/OpenXR-SDK.git GIT_TAG release-1.1.43 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(openxr_headers)
add_library(gxr_performance_boost SHARED src/layer.cpp)
target_link_libraries(gxr_performance_boost PRIVATE OpenXR::headers)
if(ANDROID)
  target_link_libraries(gxr_performance_boost PRIVATE log)
  target_link_options(gxr_performance_boost PRIVATE -Wl,-z,max-page-size=16384 -Wl,--no-undefined -Wl,-s)
endif()
target_compile_options(gxr_performance_boost PRIVATE -Wall -Wextra -Werror)
set_target_properties(gxr_performance_boost PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
option(GXR_BOOST_TESTS "Build host mock-runtime lifecycle test" OFF)
if(GXR_BOOST_TESTS AND NOT ANDROID)
  enable_testing()
  add_executable(boost_lifecycle tests/lifecycle.cpp)
  target_link_libraries(boost_lifecycle PRIVATE OpenXR::headers)
  add_test(NAME boost_lifecycle COMMAND boost_lifecycle)
endif()
