cmake_minimum_required(VERSION 3.22.1)
project(ebpatcher_whisper LANGUAGES C CXX)

if(NOT ANDROID OR NOT ANDROID_ABI MATCHES "^(arm64-v8a|x86_64)$")
    message(FATAL_ERROR "The isolated engine supports Android arm64-v8a and x86_64 only")
endif()
if(NOT DEFINED WHISPER_SOURCE_DIR)
    message(FATAL_ERROR "Use the pinned prepareWhisperSources Gradle task")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(EBPATCHER_VULKAN "Build the optional Vulkan JNI payload" OFF)
if(EBPATCHER_VULKAN)
    if(NOT EXISTS "${Vulkan_GLSLC_EXECUTABLE}" OR NOT EXISTS "${Vulkan_INCLUDE_DIR}/vulkan/vulkan.hpp")
        message(FATAL_ERROR "Vulkan requires a host glslc and Vulkan SDK headers including Vulkan-Hpp")
    endif()
    # Use NDK's Android loader, never the host SDK's libvulkan.
    find_library(ANDROID_VULKAN_LOADER vulkan REQUIRED)
    set(Vulkan_LIBRARY "${ANDROID_VULKAN_LOADER}" CACHE FILEPATH "" FORCE)
    set(WHISPER_JNI_TARGET ebpatcher_whisper_vulkan)
    find_package(SPIRV-Headers CONFIG REQUIRED)
else()
    set(WHISPER_JNI_TARGET ebpatcher_whisper)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(WHISPER_COREML OFF CACHE BOOL "" FORCE)
set(WHISPER_OPENVINO OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ${EBPATCHER_VULKAN} CACHE BOOL "" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_BMI2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH armv8-a CACHE STRING "" FORCE)
add_compile_options("-ffile-prefix-map=${WHISPER_SOURCE_DIR}=whisper.cpp"
                    "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=ebpatcher-whisper")
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper)
if(EBPATCHER_VULKAN)
    # Upstream discovers the package but does not propagate its include directory.
    target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
endif()
add_library(${WHISPER_JNI_TARGET} SHARED whisper_jni.cpp)
target_compile_definitions(${WHISPER_JNI_TARGET} PRIVATE EBPATCHER_VULKAN=$<BOOL:${EBPATCHER_VULKAN}>)
set_target_properties(${WHISPER_JNI_TARGET} PROPERTIES CXX_VISIBILITY_PRESET hidden
                      VISIBILITY_INLINES_HIDDEN YES)
target_link_libraries(${WHISPER_JNI_TARGET} PRIVATE whisper)
# Both variants contain private static ggml/Whisper copies; never interpose their globals.
target_link_options(${WHISPER_JNI_TARGET} PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
