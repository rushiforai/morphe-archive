cmake_minimum_required(VERSION 3.22)
project(gxr_foveal_canvas LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${OPENXR_SDK_SOURCE_DIR}/include/openxr/openxr.h")
    message(FATAL_ERROR "Supply OPENXR_SDK_SOURCE_DIR")
endif()
add_library(gxr_foveal_canvas SHARED src/foveal_canvas_layer.cpp)
target_include_directories(gxr_foveal_canvas PRIVATE "${OPENXR_SDK_SOURCE_DIR}/include")
target_compile_definitions(gxr_foveal_canvas PRIVATE XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES)
target_compile_options(gxr_foveal_canvas PRIVATE -O2 -Wall -Wextra -Werror -Wno-missing-field-initializers -ffunction-sections -fdata-sections)
target_link_options(gxr_foveal_canvas PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
target_link_libraries(gxr_foveal_canvas PRIVATE log EGL GLESv3)
set_target_properties(gxr_foveal_canvas PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
